TABLA NUEVA: manual_income
───────────────────────────

Columns:
  id           INTEGER PRIMARY KEY
  evento_id    INTEGER (FK a eventos)
  concepto     TEXT (ej: "Venta puerta")
  monto        REAL (monto en dinero)
  descripcion  TEXT (opcional, detalles)
  creado_por   INTEGER (user_id who added it)
  created_at   DATETIME (timestamp)
  updated_at   DATETIME (timestamp)

Indexes:
  idx_manual_income_evento ON (evento_id)
